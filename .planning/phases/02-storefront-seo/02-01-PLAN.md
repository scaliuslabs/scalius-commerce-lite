```typescript
// --- SEO: store name, Organization JSON-LD, WebSite JSON-LD (per SEO-05, SEO-06) ---
const storeName = footerData.copyrightText || headerData.logo.alt || "Store";
const storefrontUrl = getRuntimeStorefrontUrl();

// Organization JSON-LD — only emit when we have a logo (per SEO-05)
const logoUrl = headerData.logo.src ? resolveMediaUrl(headerData.logo.src) : null;
const socialUrls = (footerData.social || []).map((s: { url?: string }) => s.url).filter(Boolean);
const orgJsonLd = logoUrl ? JSON.stringify({
  "@context": "https://schema.org",
  "@type": "Organization",
  "name": storeName,
  "url": storefrontUrl,
  "logo": { "@type": "ImageObject", "url": logoUrl },
  ...(socialUrls.length > 0 ? { "sameAs": socialUrls } : {}),
}) : null;

// WebSite JSON-LD with SearchAction for Google Sitelinks Search Box (per SEO-06)
const websiteJsonLd = storefrontUrl ? JSON.stringify({
  "@context": "https://schema.org",
  "@type": "WebSite",
  "name": storeName,
  "url": storefrontUrl,
  "potentialAction": {
    "@type": "SearchAction",
    "target": {
      "@type": "EntryPoint",
      "urlTemplate": `${storefrontUrl}/search?q={search_term_string}`
    },
    "query-input": "required name=search_term_string"
  }
}) : null;
```
