```typescript
export function sanitizeFtsQuery(input: string): string {
  const cleaned = input.replace(FTS5_SPECIAL_CHARS, " ").trim();
  if (!cleaned) return "";
  const tokens = cleaned.split(/\s+/).filter((t) => t.length > 0).map((t) => `${t}*`);
  return tokens.length > 0 ? tokens.join(" ") : "";
}

export function ftsMatch(
  ftsTable: FtsTable,
  sourceTable: SourceTable,
  query: string,
): SQL | undefined;
```

If the migration FAILS with an error about the `categories` option on `unicode61` (e.g., "unknown tokenizer option", "unrecognized token", or any SQLite error on the CREATE VIRTUAL TABLE statement):

  1. Rewrite `0031_bengali_fts5_tokenizer.sql` to use the trigram tokenizer fallback. Replace every occurrence of:
     ```
     tokenize = "unicode61 categories 'L* N* Co Mc Mn' remove_diacritics 2"
     ```
     with:
     ```
     tokenize = "trigram"
     ```
  2. Reset the local database: `pnpm dev:reset`
  3. Re-run: `pnpm db:migrate:local`
  4. Note in the SUMMARY that trigram fallback was used and why.
